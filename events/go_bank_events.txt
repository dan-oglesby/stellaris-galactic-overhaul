namespace = go_bank

# ============================================================================
# go_bank.1 - the Treasury Reserve panel window (diplomatic custom_gui).
# Opened by the go_open_bank_edict edict (primary) or decision_go_open_bank
# (capital fallback). Root = the player country. Saves a global event target
# (go_bank_player) that the readout loc tokens resolve against. The readouts read
# LIVE resources / persisted rate on each GUI redraw (see
# common/script_values/go_bank_script_values.txt), so the panel self-refreshes
# after a deposit/withdraw without re-firing this event.
# ============================================================================
country_event = {
	id = go_bank.1
	is_triggered_only = yes
	diplomatic = yes
	custom_gui = "go_bank_window"
	title = "go_bank.1.title"
	desc  = "go_bank.1.desc"
	trigger = { is_ai = no }

	immediate = {
		save_global_event_target_as = go_bank_player
		if = { limit = { go_debug_on = yes } log = "GALACTIC_OVERHAUL: reserve panel opened" }
	}

	option = { name = "go_bank.1.close" default_hide_option = yes }
}
