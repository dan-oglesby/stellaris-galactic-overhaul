# Values behind the Treasury Reserve panel readouts.
#
# WHY THESE READ LIVE STATE: re-firing go_bank.1 from a button does NOT repaint an
# already-open custom_gui window (it just queues a second one). The GUI DOES
# re-evaluate scripted_loc/script_value tokens on every redraw, so reading live
# resources here is what makes the panel self-update after a deposit/withdraw.
#
# SYNTAX NOTE: script values do not support `value =` or `if = { limit = ... }`.
# Live resource reads use vanilla's parameterised get_resource_amount
# (07_script_values_shroud.txt:599); persisted-variable reads use
# `modifier = { <op> <gating trigger> }`.

# live Banked Credits balance
go_disp_banked_sv = {
	base = 0
	add = value:get_resource_amount|RESOURCE|go_banked_credits|PERCENTAGE|1|
}

# live liquid Trade on hand (what's NOT in the Reserve)
go_disp_liquid_sv = {
	base = 0
	add = value:get_resource_amount|RESOURCE|trade|PERCENTAGE|1|
}

# stated monthly interest rate from the last tick, as a percentage
go_disp_interest_sv = {
	base = 0
	modifier = {
		add = go_last_rate
		is_variable_set = go_last_rate
	}
	mult = 100
}

# interest actually paid last month (persisted by galactic_overhaul.1)
go_disp_interest_amt_sv = {
	base = 0
	modifier = {
		add = go_last_interest
		is_variable_set = go_last_interest
	}
}
