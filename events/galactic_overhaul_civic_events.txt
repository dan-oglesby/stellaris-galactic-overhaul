namespace = galactic_overhaul_civic

# ============================================================================
# The megacorp founding civics (civic_go_banking_charter / civic_go_market_authority)
# were REMOVED in v0.26.0. They instantly founded a galaxy-wide institution for every
# empire from turn one, which is a sweeping galaxy change rather than automation - and
# an AI rolling one silently founded the Galactic Market in someone else's game.
# The Galactic Bank is now chartered solely through the Galactic Community resolution,
# and the Galactic Market through vanilla's own process.
# Events .1 (founding), .2 and .3 (the galaxy-wide popups) went with them.
#
# What remains is the independent "no AI megacorps" rule. Debug tag: GALACTIC_OVERHAUL.
# ============================================================================

# .4 - "No AI megacorps" rule (unconditional): at game start, convert any AI
# megacorp into a regular oligarchic empire. The player (is_ai = no) is never
# touched, so you can still play a megacorp. remove_invalid_civics drops the now-
# invalid corporate civics; civics = random rolls a valid replacement set.
country_event = {
	id = galactic_overhaul_civic.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		is_country_type = default
		has_authority = auth_corporate
	}

	immediate = {
		change_government = {
			authority = auth_oligarchic
			civics = random
			remove_invalid_civics = yes
			cooldown = no
		}
		if = { limit = { go_debug_on = yes } log = "GALACTIC_OVERHAUL: converted AI megacorp [This.GetName] to oligarchic (no-AI-megacorps rule)" }
	}
}
