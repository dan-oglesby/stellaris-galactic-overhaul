# ============================================================================
# Treasury Reserve (Galactic Banking) - tunable constants
# ============================================================================
# The Reserve pays a FLAT, STATED monthly interest rate on the Banked Credits
# balance - no hidden market model. These percentages are what the player sees in
# the policy tooltip and the Reserve panel, so keep the loc in sync if you retune.

# Flat monthly interest rates by policy tier.
@go_rate_conservative  = 0.003   # 0.3%/month, no risk of loss
@go_rate_standard      = 0.006   # 0.6%/month, ~8% monthly chance of a loss
@go_rate_aggressive    = 0.012   # 1.2%/month, ~15% monthly chance of a larger loss
# Extra flat rate while your empire is the Galactic Bank host.
@go_rate_host_bonus    = 0.0025  # +0.25%/month

# --- Risk of loss (tiers above Conservative) ---
# On a bad month, lose this fraction of the current balance. The monthly CHANCE is
# set in galactic_overhaul.1's random_list (Standard ~8%, Aggressive ~15%).
@go_bank_loss_std = 0.05   # Standard: -5% of balance on a loss
@go_bank_loss_agg = 0.10   # Aggressive: -10% of balance on a loss
