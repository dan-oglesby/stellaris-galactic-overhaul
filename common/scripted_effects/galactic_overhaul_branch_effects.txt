# ============================================================================
# Branch Office Automation - holding construction.
# Scope when called: this = the HOST colony (holds root's branch office),
#                    root = the megacorp. Caller has already confirmed a free
#                    branch-office building slot and that root can afford a holding.
#
# NOTE: presence checks use has_holding = { holding = X owner = root }, NOT has_building.
# Branch-office buildings are owner_type = corporate and the engine rejects has_building on
# them outright ("Use has_holding instead", ~20 errors per sweep in the 2026-07-26 playtest),
# which silently broke every "don't rebuild what's already here" guard in this ladder.
#
# add_holding = { holding = X owner = root } is the owner-aware build (add_building
# would misattribute the holding to the host empire). It BYPASSES the building's own
# potential/allow checks, so this ladder does the gating itself.
# ============================================================================

# Build one holding + pay the real cost + count it against the monthly budget.
# NOTE: $HOLDING$ is a parse-time param (safe in a log); scope macros like
# [This.x] are NOT rendered in scripted_effect logs, so none are used here.
go_bo_do_build_holding = {
	# ATTEMPT MARKER. add_holding bypasses the building's own potential/allow checks, so it
	# can silently refuse (or produce something has_holding does not report) - and we charge
	# unconditionally. In the 2026-07-30 playtest that meant building_underground_chemists
	# was "built" three times on the SAME office, billing 25 influence + 500 minerals each
	# time, because the has_holding guard in the ladder never saw it and re-picked it every
	# sweep. A timed per-planet, per-holding flag caps that at one attempt per window
	# instead of one per month, whether or not has_holding reports the result immediately.
	if = {
		limit = { has_planet_flag = go_bo_tried_$HOLDING$ }
		if = { limit = { root = { go_debug_on = yes } } log = "GALACTIC_OVERHAUL_BRANCH: skipping $HOLDING$ - attempted here recently" }
	}
	else = {
	set_timed_planet_flag = { flag = go_bo_tried_$HOLDING$ days = @go_bo_retry_days }
	add_holding = { holding = $HOLDING$ owner = root }
	root = {
		add_resource = { influence = @go_bo_holding_influence_neg minerals = @go_bo_holding_minerals_neg }
		change_variable = { which = go_bo_hbudget value = -1 }
		# Advance the diversity rotation so the NEXT office starts from a different
		# candidate. Without this the ladder is a fixed if/else_if chain and every
		# office ends up with whatever matches first (the 2026-07-28 playtest built
		# building_private_shipyards 58 times in a row).
		change_variable = { which = go_bo_rotate value = 1 }
		if = {
			limit = { check_variable = { which = go_bo_rotate value >= @go_bo_rotate_slots } }
			set_variable = { which = go_bo_rotate value = 0 }
		}
		if = { limit = { go_debug_on = yes } log = "GALACTIC_OVERHAUL_BRANCH: built $HOLDING$ (macro-free: scripted_effect logs don't render scope tokens)" }
	}
	}
}

# ---------------------------------------------------------------------------
# DISPATCHER: criminal syndicates cannot use the legit building set at all.
# Every building in the legit ladder has potential = { branch_office_owner = {
# is_legit_or_legal_criminal = yes } }, which is FALSE for a crime corp operating
# illegally - and add_holding BYPASSES potential, so before this fix the automation
# happily created legit holdings on a criminal branch office that could never
# function. (2026-07-28 playtest: 58 holdings "built", all invalid -> "no buildings".)
#
# KNOWN LIMITATION: this branches on is_criminal_syndicate, not on whether the office
# has been LEGALISED on that particular planet. A legalised crime corp uses the legit
# set in vanilla; here it would still get criminal holdings. Vanilla's own
# is_legit_or_legal_criminal trigger cannot be called from this effect because it
# references root.owner, and our root is the megacorp rather than the planet.
# ---------------------------------------------------------------------------
go_bo_build_best_holding = {
	# seed the diversity rotation on first use
	root = {
		if = {
			limit = { NOT = { is_variable_set = go_bo_rotate } }
			set_variable = { which = go_bo_rotate value = 0 }
		}
	}
	if = {
		limit = { root = { is_criminal_syndicate = yes } }
		go_bo_build_best_holding_criminal = yes
	}
	else = {
		go_bo_build_best_holding_legit = yes
	}
}

	else_if = {
		limit = { NOT = { has_holding = { holding = building_commercial_forum owner = root } } }
		go_bo_do_build_holding = { HOLDING = building_commercial_forum }

	else_if = {
		limit = { NOT = { has_holding = { holding = building_private_research_initiative owner = root } } }

		limit = { NOT = { has_holding = { holding = building_military_contractors owner = root } } }
