namespace = galactic_overhaul_gsm

# ============================================================================
# Galactic Stock Market compatibility (Workshop 3813910097, "Galactic Stock Market" v1.0.0).
#
# Policy go_dividend_reinvestment decides what happens to the dividends Galactic Stock Market
# pays quarterly, as trade, to every empire holding shares (gsm_company_pay_dividend):
#   go_drip_off    take them as trade (GSM's own behaviour; nothing runs here)
#   go_drip_on     reinvest what each company paid in that same company   (flag go_drip_active)
#   go_drip_index  reinvest all of it towards a market-cap-weighted portfolio (flag go_drip_index)
#   go_drip_desk   quarterly value trading against fundamental value     (flag go_drip_desk)
# Purchases go through GSM's own quote/buy API - the same path its AI uses
# (gsm_ai_buy_within_budget) - so the 1 % fee, price impact, foreign-ownership limits, contact
# rules and affordability all apply exactly as for a manual purchase. Nothing of GSM is overridden.
#
# Design notes (read against GSM's files):
# - Timing. GSM runs its whole market once a month from on_monthly_pulse (gsm_core.100) and
#   leaves event_target:gsm_exchange.gsm_t_qend = 1 after a quarter-end tick until the next tick.
#   The order of GSM's pulse and this one is not guaranteed, but either way exactly one of our
#   monthly passes sees gsm_t_qend = 1 after each payout. go_drip_last records the exchange's
#   month counter (gsm_month) so a pass can never run twice for the same payout.
# - Amount. Each holder was paid holding * total / shares; gsm_div_ps keeps that quarter's
#   regular + special dividend per share, so holding * gsm_div_ps is the payout. GSM notes that
#   script variables truncate to 5 decimals, so for companies with very many shares this can read
#   slightly low - reinvestment errs on the side of spending less than was paid, never more.
# - Whole shares only (GSM has no fractional shares). What does not buy a whole share stays in
#   the trade stockpile.
# - Nesting. The purchase itself runs in galactic_overhaul_gsm.2, fired synchronously, which GSM
#   uses as a "fresh nesting root" for the same reason: its quote effect nests deeply.
# Debug tag: GALACTIC_OVERHAUL_GSM.
# ============================================================================
country_event = {
	id = galactic_overhaul_gsm.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		# PERF: every reinvesting option has ai_weight = 0, so AI empires never hold these flags;
		# is_ai short-circuits the pulse for them.
		is_ai = no
		OR = {
			has_policy_flag = go_drip_active
			has_policy_flag = go_drip_index
			has_policy_flag = go_drip_desk
		}
		has_global_flag = gsm_initialized
		exists = event_target:gsm_exchange
		event_target:gsm_exchange = {
			check_variable = { which = gsm_t_qend value = 1 }
		}
		gsm_has_iid = yes
		NOT = {
			check_variable = { which = go_drip_last value = event_target:gsm_exchange.gsm_month }
		}
	}

	immediate = {
		set_variable = { which = go_drip_last value = event_target:gsm_exchange.gsm_month }
		set_variable = { which = go_drip_spent_q value = 0 }
		set_variable = { which = go_drip_shares_q value = 0 }
		save_event_target_as = go_drip_holder

		# ---------------------------------------------------------------- reinvest in payer
		if = {
			limit = { has_policy_flag = go_drip_active }
			every_country = {
				limit = {
					is_country_type = gsm_company
					check_variable = { which = gsm_div_ps value > 0 }
					check_variable = { which = gsm_price value > 0 }
					check_variable = { which = gsm_shares value > 0 }
				}
				save_event_target_as = go_drip_co
				if = {
					limit = {
						event_target:go_drip_holder = { has_country_flag = gsm_holds@event_target:go_drip_co }
					}
					# this holder's position in the company (GSM's generated per-investor store)
					set_variable = { which = gsm_t_iid value = event_target:go_drip_holder.gsm_iid }
					gsm_gen_hold_get = yes
					save_event_target_as = gsm_t_tc
					event_target:go_drip_holder = {
						# budget = holding x dividend per share = what this company just paid us
						set_variable = { which = go_drip_budget value = event_target:go_drip_co.gsm_t_hold }
						multiply_variable = { which = go_drip_budget value = event_target:go_drip_co.gsm_div_ps }
						country_event = { id = galactic_overhaul_gsm.2 }
					}
				}
			}
		}

		# ---------------------------------------------------------------- index portfolio
		# Target: each listed company held in proportion to its market cap (the weighting of GSM's
		# Galactic Composite Index). New money only - nothing is sold - so each quarter the whole
		# payout goes to the companies furthest BELOW their index weight, largest gap first.
		else_if = {
			limit = { has_policy_flag = go_drip_index }
			set_variable = { which = go_drip_budget value = 0 }
			set_variable = { which = go_ix_pv value = 0 }
			set_variable = { which = go_ix_mt value = 0 }
			# pass 1: total payout, current portfolio value, and total market cap
			every_country = {
				limit = {
					gsm_is_listed_company = yes
					check_variable = { which = gsm_price value > 0 }
					check_variable = { which = gsm_shares value > 0 }
				}
				save_event_target_as = go_drip_co
				remove_country_flag = go_ix_tried
				set_variable = { which = go_ix_val value = 0 }
				if = {
					limit = {
						event_target:go_drip_holder = { has_country_flag = gsm_holds@event_target:go_drip_co }
					}
					set_variable = { which = gsm_t_iid value = event_target:go_drip_holder.gsm_iid }
					gsm_gen_hold_get = yes
					set_variable = { which = go_ix_val value = gsm_t_hold }
					multiply_variable = { which = go_ix_val value = gsm_price }
					event_target:go_drip_holder = {
						change_variable = { which = go_ix_pv value = event_target:go_drip_co.go_ix_val }
						if = {
							limit = { event_target:go_drip_co = { check_variable = { which = gsm_div_ps value > 0 } } }
							set_variable = { which = go_ix_pay value = event_target:go_drip_co.gsm_t_hold }
							multiply_variable = { which = go_ix_pay value = event_target:go_drip_co.gsm_div_ps }
							change_variable = { which = go_drip_budget value = go_ix_pay }
						}
					}
				}
				event_target:go_drip_holder = {
					change_variable = { which = go_ix_mt value = event_target:go_drip_co.gsm_mcap_k }
				}
			}
			# pass 2: gap = target value - held value, target = mcap share x (portfolio + payout)
			if = {
				limit = {
					check_variable = { which = go_drip_budget value >= 1 }
					check_variable = { which = go_ix_mt value > 0 }
				}
				set_variable = { which = go_ix_v value = go_ix_pv }
				change_variable = { which = go_ix_v value = go_drip_budget }
				every_country = {
					limit = {
						gsm_is_listed_company = yes
						check_variable = { which = gsm_price value > 0 }
						check_variable = { which = gsm_shares value > 0 }
					}
					set_variable = { which = go_ix_gap value = gsm_mcap_k }
					divide_variable = { which = go_ix_gap value = event_target:go_drip_holder.go_ix_mt }
					multiply_variable = { which = go_ix_gap value = event_target:go_drip_holder.go_ix_v }
					subtract_variable = { which = go_ix_gap value = go_ix_val }
				}
				# pass 3: up to six purchases, each into the most underweight company not yet tried
				# (a company the exchange will not sell to us - no contact, foreign cap - is skipped)
				set_variable = { which = go_ix_left value = go_drip_budget }
				while = {
					count = 6
					if = {
						limit = { check_variable = { which = go_ix_left value >= 1 } }
						ordered_country = {
							limit = {
								gsm_is_listed_company = yes
								check_variable = { which = go_ix_gap value > 0 }
								NOT = { has_country_flag = go_ix_tried }
							}
							position = 0
							order_by = go_ix_gap
							set_country_flag = go_ix_tried
							save_event_target_as = gsm_t_tc
							save_event_target_as = go_drip_co
							event_target:go_drip_holder = {
								# spend min(gap, what is left of the payout)
								set_variable = { which = go_drip_budget value = go_ix_left }
								if = {
									limit = {
										check_variable = { which = go_drip_budget value > event_target:go_drip_co.go_ix_gap }
									}
									set_variable = { which = go_drip_budget value = event_target:go_drip_co.go_ix_gap }
								}
								set_variable = { which = go_drip_before value = go_drip_spent_q }
								country_event = { id = galactic_overhaul_gsm.2 }
								subtract_variable = { which = go_ix_left value = go_drip_spent_q }
								change_variable = { which = go_ix_left value = go_drip_before }
							}
						}
					}
				}
			}
			every_country = {
				limit = { has_country_flag = go_ix_tried }
				remove_country_flag = go_ix_tried
			}
		}

		# ---------------------------------------------------------------- active desk
		# Expected-value trading, once a quarter (Dan, 2026-10-10: "any trade where the expected value
		# beats costs, filling from high to low value"). The model is GSM's own price rule, using only
		# figures GSM shows its players (target price, trailing dividends) and its own constants:
		#   - each month a price closes @gsm_price_lambda of the gap to its target price, so over a
		#     quarter it closes conv = 1 - (1 - lambda)^3 of it (0.657 at lambda 0.3);
		#   - a quarter's dividend is about a quarter of the trailing twelve months' (gsm_div_ttm);
		#   - every trade pays @gsm_trade_fee, and moving a fraction f of a company's shares moves its
		#     price by @gsm_impact_k x f (the trade itself executes at half of that move).
		# Expected quarterly return of holding a share, as a fraction of its price:
		#   h = conv x (target / price - 1) + (dividends ttm / 4) / price
		# Buying a fraction f of the company and assuming the position is sold again after one quarter
		# (the conservative case) works out to an expected profit, per share, of
		#   edge - k x conv x f,   with  edge = h - 2 x fee
		# so a buy is worth making when edge > 0, and the most worth buying is f = edge / (2 k conv),
		# where the extra price impact cancels the edge. Companies are filled best edge first.
		# Selling a held position now instead of holding it a quarter is worth it when
		#   -h - fee > 0   (the expected loss exceeds the exit fee), up to f = (-h - fee) / k.
		# Unchanged rules: distressed holdings are always sold; companies of our own empire are never
		# sold; budget @go_dk_spend_share of trade above @go_dk_reserve; no company above
		# @go_dk_cap_share of the portfolio. Dividends stay as trade and feed the budget.
		else_if = {
			limit = { has_policy_flag = go_drip_desk }
			set_variable = { which = go_dk_sold_q value = 0 }
			set_variable = { which = go_dk_sold_n value = 0 }
			set_variable = { which = go_dk_buys value = 0 }
			# model constants, from GSM's own settings
			set_variable = { which = go_dk_pw value = 1 }
			subtract_variable = { which = go_dk_pw value = @gsm_price_lambda }
			set_variable = { which = go_dk_conv value = go_dk_pw }
			multiply_variable = { which = go_dk_conv value = go_dk_pw }
			multiply_variable = { which = go_dk_conv value = go_dk_pw }
			multiply_variable = { which = go_dk_conv value = -1 }
			change_variable = { which = go_dk_conv value = 1 }
			set_variable = { which = go_dk_fee2 value = @gsm_trade_fee }
			multiply_variable = { which = go_dk_fee2 value = 2 }
			set_variable = { which = go_dk_kc2 value = @gsm_impact_k }
			multiply_variable = { which = go_dk_kc2 value = go_dk_conv }
			multiply_variable = { which = go_dk_kc2 value = 2 }

			# pass 1: expected quarterly return h for every listed company with a target price
			every_country = {
				limit = {
					gsm_is_live_company = yes
					check_variable = { which = gsm_price value > 0 }
					check_variable = { which = gsm_shares value > 0 }
				}
				set_variable = { which = go_dk_ok value = 0 }
				if = {
					limit = { check_variable = { which = gsm_target value > 0 } }
					set_variable = { which = go_dk_ok value = 1 }
					set_variable = { which = go_dk_h value = gsm_target }
					divide_variable = { which = go_dk_h value = gsm_price }
					change_variable = { which = go_dk_h value = -1 }
					multiply_variable = { which = go_dk_h value = event_target:go_drip_holder.go_dk_conv }
					set_variable = { which = go_dk_y value = 0 }
					if = {
						limit = { check_variable = { which = gsm_div_ttm value > 0 } }
						set_variable = { which = go_dk_y value = gsm_div_ttm }
						divide_variable = { which = go_dk_y value = 4 }
						divide_variable = { which = go_dk_y value = gsm_price }
					}
					change_variable = { which = go_dk_h value = go_dk_y }
				}
			}

			# pass 2: sells - distressed, or expected to lose more than the exit fee (never our own)
			every_country = {
				limit = {
					gsm_is_live_company = yes
					check_variable = { which = gsm_shares value > 0 }
					NOT = { check_variable = { which = gsm_home_iid value = event_target:go_drip_holder.gsm_iid } }
				}
				save_event_target_as = go_drip_co
				if = {
					limit = {
						event_target:go_drip_holder = { has_country_flag = gsm_holds@event_target:go_drip_co }
						OR = {
							has_country_flag = gsm_distress
							AND = {
								check_variable = { which = go_dk_ok value = 1 }
								check_variable_arithmetic = {
									which = go_dk_h
									add = @gsm_trade_fee
									value < 0
								}
							}
						}
					}
					save_event_target_as = gsm_t_tc
					event_target:go_drip_holder = {
						# fraction of the company to sell: everything when distressed, else (-h - fee) / k
						set_variable = { which = gsm_t_lotf value = 1 }
						if = {
							limit = { NOT = { event_target:go_drip_co = { has_country_flag = gsm_distress } } }
							set_variable = { which = gsm_t_lotf value = event_target:go_drip_co.go_dk_h }
							change_variable = { which = gsm_t_lotf value = @gsm_trade_fee }
							multiply_variable = { which = gsm_t_lotf value = -1 }
							divide_variable = { which = gsm_t_lotf value = @gsm_impact_k }
						}
						country_event = { id = galactic_overhaul_gsm.3 }   # GSM's quote caps it at our holding
					}
				}
			}

			# pass 3: budget, portfolio value, and per company: buy edge, size worth buying, room
			export_resource_stockpile_to_variable = { resource = trade variable = go_dk_stock }
			set_variable = { which = go_dk_left value = go_dk_stock }
			subtract_variable = { which = go_dk_left value = @go_dk_reserve }
			multiply_variable = { which = go_dk_left value = @go_dk_spend_share }
			set_variable = { which = go_dk_pv value = 0 }
			every_country = {
				limit = {
					gsm_is_listed_company = yes
					check_variable = { which = gsm_price value > 0 }
					check_variable = { which = gsm_shares value > 0 }
				}
				save_event_target_as = go_drip_co
				remove_country_flag = go_dk_tried
				set_variable = { which = go_dk_val value = 0 }
				if = {
					limit = {
						event_target:go_drip_holder = { has_country_flag = gsm_holds@event_target:go_drip_co }
					}
					set_variable = { which = gsm_t_iid value = event_target:go_drip_holder.gsm_iid }
					gsm_gen_hold_get = yes
					set_variable = { which = go_dk_val value = gsm_t_hold }
					multiply_variable = { which = go_dk_val value = gsm_price }
					event_target:go_drip_holder = {
						change_variable = { which = go_dk_pv value = event_target:go_drip_co.go_dk_val }
					}
				}
				# edge = h - 2 x fee (-1 when there is no target price: never a buy)
				set_variable = { which = go_dk_e value = -1 }
				set_variable = { which = go_dk_want value = 0 }
				if = {
					limit = { check_variable = { which = go_dk_ok value = 1 } }
					set_variable = { which = go_dk_e value = go_dk_h }
					subtract_variable = { which = go_dk_e value = event_target:go_drip_holder.go_dk_fee2 }
					if = {
						limit = { check_variable = { which = go_dk_e value > 0 } }
						# f = edge / (2 k conv), at most @go_dk_max_lot of the company; spend = f x shares x price x (1 + fee)
						set_variable = { which = go_dk_f value = go_dk_e }
						divide_variable = { which = go_dk_f value = event_target:go_drip_holder.go_dk_kc2 }
						if = {
							limit = { check_variable = { which = go_dk_f value > @go_dk_max_lot } }
							set_variable = { which = go_dk_f value = @go_dk_max_lot }
						}
						set_variable = { which = go_dk_want value = go_dk_f }
						multiply_variable = { which = go_dk_want value = gsm_shares }
						multiply_variable = { which = go_dk_want value = gsm_price }
						multiply_variable = { which = go_dk_want value = 1.01 }
					}
				}
			}
			if = {
				limit = { check_variable = { which = go_dk_left value >= 1 } }
				# cap per company = share x (portfolio + budget); room = cap - held value
				set_variable = { which = go_dk_cap value = go_dk_pv }
				change_variable = { which = go_dk_cap value = go_dk_left }
				multiply_variable = { which = go_dk_cap value = @go_dk_cap_share }
				every_country = {
					limit = {
						gsm_is_listed_company = yes
						check_variable = { which = gsm_price value > 0 }
						check_variable = { which = gsm_shares value > 0 }
					}
					set_variable = { which = go_dk_room value = event_target:go_drip_holder.go_dk_cap }
					subtract_variable = { which = go_dk_room value = go_dk_val }
				}
				# pass 4: fill best edge first; skip companies the exchange will not sell to us
				while = {
					count = @go_dk_max_buys
					if = {
						limit = { check_variable = { which = go_dk_left value >= 1 } }
						ordered_country = {
							limit = {
								gsm_is_listed_company = yes
								NOT = { has_country_flag = gsm_distress }
								NOT = { has_country_flag = go_dk_tried }
								check_variable = { which = go_dk_e value > 0 }
								check_variable = { which = go_dk_want value >= 1 }
								check_variable = { which = go_dk_room value >= 1 }
							}
							position = 0
							order_by = go_dk_e
							set_country_flag = go_dk_tried
							save_event_target_as = gsm_t_tc
							save_event_target_as = go_drip_co
							event_target:go_drip_holder = {
								# spend min(what is left, room under the cap, the size worth buying)
								set_variable = { which = go_drip_budget value = go_dk_left }
								if = {
									limit = { check_variable = { which = go_drip_budget value > event_target:go_drip_co.go_dk_room } }
									set_variable = { which = go_drip_budget value = event_target:go_drip_co.go_dk_room }
								}
								if = {
									limit = { check_variable = { which = go_drip_budget value > event_target:go_drip_co.go_dk_want } }
									set_variable = { which = go_drip_budget value = event_target:go_drip_co.go_dk_want }
								}
								set_variable = { which = go_drip_before value = go_drip_spent_q }
								country_event = { id = galactic_overhaul_gsm.2 }
								subtract_variable = { which = go_dk_left value = go_drip_spent_q }
								change_variable = { which = go_dk_left value = go_drip_before }
								if = {
									limit = { check_variable = { which = go_drip_spent_q value > go_drip_before } }
									change_variable = { which = go_dk_buys value = 1 }
								}
							}
						}
					}
				}
			}
			every_country = {
				limit = { has_country_flag = go_dk_tried }
				remove_country_flag = go_dk_tried
			}
			if = {
				limit = { go_debug_on = yes }
				log = "GALACTIC_OVERHAUL_GSM: [This.GetName] desk sold [This.go_dk_sold_n] shares for [This.go_dk_sold_q] trade; bought into [This.go_dk_buys] companies (budget [This.go_dk_left] left)"
			}
		}

		# start the lifetime total at 0 so the log never prints an empty value
		if = {
			limit = { NOT = { is_variable_set = go_drip_total } }
			set_variable = { which = go_drip_total value = 0 }
		}
		if = {
			limit = { check_variable = { which = go_drip_spent_q value > 0 } }
			change_variable = { which = go_drip_total value = go_drip_spent_q }
		}
		if = {
			limit = { go_debug_on = yes }
			log = "GALACTIC_OVERHAUL_GSM: [This.GetName] reinvested [This.go_drip_spent_q] trade in [This.go_drip_shares_q] shares (lifetime [This.go_drip_total])"
		}
	}
}

# ============================================================================
# galactic_overhaul_gsm.2 - one budgeted purchase (fired synchronously from .1).
# Holder country scope; company in event_target:gsm_t_tc (also event_target:go_drip_co); spends at
# most go_drip_budget trade. Adds what it spent to go_drip_spent_q / go_drip_shares_q.
# ============================================================================
country_event = {
	id = galactic_overhaul_gsm.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# estimated whole shares at price + 1 % fee; nothing to do below one share
		set_variable = { which = go_drip_est value = go_drip_budget }
		divide_variable = { which = go_drip_est value = event_target:gsm_t_tc.gsm_price }
		divide_variable = { which = go_drip_est value = 1.01 }
		if = {
			limit = { check_variable = { which = go_drip_est value >= 1 } }
			# GSM sizes a buy as a fraction of all shares (gsm_t_lotf)
			set_variable = { which = gsm_t_lotf value = go_drip_est }
			divide_variable = { which = gsm_t_lotf value = event_target:gsm_t_tc.gsm_shares }
			gsm_trade_quote_buy = yes
			# price impact can push the quote over budget: shrink the lot once and requote
			if = {
				limit = {
					check_variable = { which = gsm_t_trade_q value > 1 }
					check_variable = { which = gsm_t_trade_cost value > go_drip_budget }
				}
				set_variable = { which = go_drip_scale value = go_drip_budget }
				divide_variable = { which = go_drip_scale value = gsm_t_trade_cost }
				multiply_variable = { which = go_drip_scale value = 0.99 }

				gsm_trade_quote_buy = yes
			}
			if = {
				limit = {

					check_variable = { which = gsm_t_trade_cost value <= go_drip_budget }
				}
				gsm_trade_buy = yes

					change_variable = { which = go_drip_shares_q value = gsm_t_trade_q }
				}
			}
		}
	}
