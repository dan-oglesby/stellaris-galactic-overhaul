# "Branch Office Automation" - empire-wide megacorp policy (player QoL).
# When On, galactic_overhaul_branch.1 runs a monthly sweep that opens branch offices
# on qualifying colonies and builds the most beneficial holding on offices you own,
# paying the real influence/mineral costs. Megacorp-only (requires MegaCorp DLC).
branch_office_automation = {
	# 4.5 groups policies under collapsible category headers. Natural home plus the mod's own
	# group, so every Galactic Overhaul toggle can be found in one place. Cosmetic; save-safe.
	category = { economy galactic_overhaul }
	# NOTE: use is_megacorp (+ has_megacorp for the DLC check) - that is the pair vanilla
	# policies use (00_policies.txt:443-447). `authority = { value = auth_corporate }` is a
	# CIVIC-context trigger; no vanilla POLICY uses it, and it silently suppressed this
	# whole policy (same failure mode as is_ai in a decision potential).
	potential = {
		is_country_type = default
		has_megacorp = yes      # MegaCorp DLC present
		is_megacorp = yes       # corporate authority
	}

	allow = {
		always = yes
	}

	# Default: off (you place branch offices and holdings by hand).
	option = {
		name = "go_bo_off"
		icon = "GFX_diplomatic_stance_withdrawn"
		ai_weight = {
			weight = 100	# AI megacorps keep manual behaviour by default.
		}
	}

	option = {
		name = "go_bo_on"
		icon = "GFX_diplomatic_stance_mercantile"
		policy_flags = {
			go_branch_active
		}
		ai_weight = {
			weight = 0
		}
	}

	# Criminal Syndicates only (Dan, 2026-10-10): a crime corp needs no commercial pact, so plain
	# "on" spreads criminal offices to every eligible empire. These two keep the automation but limit
	# where it EXPANDS; holdings on offices you already have are built either way
	# (go_bo_colony_qualifies / go_bo_host_hostile in galactic_overhaul_branch_triggers.txt).
	option = {
		name = "go_bo_crime_hostile"
		icon = "GFX_diplomatic_stance_belligerent"
		potential = {
			is_criminal_syndicate = yes
		}
		policy_flags = {
			go_branch_active
			go_bo_hostile_only
		}
		ai_weight = {
			weight = 0
		}
	}

	option = {
		name = "go_bo_crime_holdings"
		icon = "GFX_diplomatic_stance_isolationist"
		potential = {
			is_criminal_syndicate = yes
		}
		policy_flags = {
			go_branch_active
			go_bo_no_new
		}
		ai_weight = {
			weight = 0
		}
	}
}
