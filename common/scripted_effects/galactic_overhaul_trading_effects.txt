# ============================================================================
# OVERRIDE of vanilla choose_galactic_market_host - picks the strongest-economy
# eligible empire (value:go_economy_score) instead of the market_rating bid.
# Loads after vanilla, so this wins (last-loaded override).
# ============================================================================
choose_galactic_market_host = {
	# PASS 1 - the vanilla path: choose among empires flagged eligible by an in-progress
	# Community market nomination. Unchanged behaviour for vanilla's own founding flow.
	ordered_country = {
		limit = {
			is_country_type = default
			has_country_flag = market_nomination_eligible
			any_owned_planet = { is_market_nominee_planet = yes }
		}
		order_by = value:go_economy_score
		position = 0
		save_global_event_target_as = market_founder
		if = {
			limit = { go_debug_on = yes }
			log = "GALACTIC_OVERHAUL: galactic market host chosen (strongest-economy override active)"
		}
		random_owned_planet = {
			limit = { is_market_nominee_planet = yes }
			save_global_event_target_as = market_founder_planet
		}
	}
	# PASS 2 - fallback for founding OUTSIDE a nomination (e.g. the megacorp Market
	# Authority civic at game start). Vanilla only sets market_nomination_eligible during
	# a Community nomination, so pass 1 matches nobody then and the market would be
	# "founded" with no host at all. Only runs if pass 1 found none, so it never changes
	# the vanilla outcome. NOTE: this pass must NOT use is_market_nominee_planet either -
	# that trigger requires market_nominee_level_* modifiers which vanilla only applies
	# during a nomination, so an earlier version of this fallback still matched nobody and
	# the 2026-07-26 playtest logged "market founding deferred" 88 times in a row. Uses the
	# capital as the host planet instead.
	if = {
		limit = { NOT = { exists = event_target:market_founder } }
		ordered_country = {
			limit = {
				is_country_type = default
				any_owned_planet = { is_capital = yes }
			}
			order_by = value:go_economy_score
			position = 0
			save_global_event_target_as = market_founder
			if = {
				limit = { go_debug_on = yes }
				log = "GALACTIC_OVERHAUL: galactic market host chosen via fallback (no nomination in progress)"
			}
			random_owned_planet = {
				limit = { is_capital = yes }
				save_global_event_target_as = market_founder_planet
			}
		}
	}
	every_playable_country = {
		limit = { has_country_flag = market_nomination_eligible }
		remove_galactic_market_modifiers = yes
	}
}

# ============================================================================
# galactic_overhaul_trade_resource - parameterized auto-trade for ONE resource.
# Call as: galactic_overhaul_trade_resource = { RESOURCE = minerals }
# ----------------------------------------------------------------------------
# SELL when a resource is going to waste: stockpile is near its CAP (>= max *
#   @go_trade_sell_cap_frac) AND income is positive (overflow being lost) ->
#   sell the surplus down to max * @go_trade_sell_keep_frac.
# BUY (restock) whenever a resource is below its reserve target (< max *
#   $TARGETFRAC$) -> buy up TOWARD that reserve, regardless of income
#   sign, capped per month and by spare trade. This keeps bursty consumers (alloys)
#   topped up instead of only reacting once they are already near empty and bleeding.
# Trades at the live fee-free market price * a simulated fee * a price-impact
# penalty that grows with accumulated trade pressure. Reads cap via
# export_resource_maximum_to_variable and income via export_resource_income_to_variable.
# ============================================================================
galactic_overhaul_trade_resource = {
	# init + monthly decay of price-impact pressure
	if = {
		limit = { NOT = { is_variable_set = go_pressure_$RESOURCE$ } }
		set_variable = { which = go_pressure_$RESOURCE$ value = 0 }
	}
	multiply_variable = { which = go_pressure_$RESOURCE$ value = @go_trade_pressure_decay }
	# action result for the caller's debug log: 0 = no action, 1 = sold, 2 = bought
	set_variable = { which = go_t_action value = 0 }
	set_variable = { which = go_t_amt value = 0 }

	# read live price, stockpile, cap, income
	export_trigger_value_to_variable = {
		trigger = market_resource_price
		parameters = { resource = $RESOURCE$ amount = 1 trade_type = not_set }
		variable = go_t_price
	}
	# Seed the opportunistic reference price on first sight of this resource. The EMA is
	# maintained EVERY month (see the update at the end), not only on months when the
	# safety desk stayed idle - otherwise the reference goes stale exactly while the desk
	# is busy, and the buy-low/sell-high band drifts away from the real market.
	if = { limit = { NOT = { is_variable_set = go_opp_mean_$RESOURCE$ } } set_variable = { which = go_opp_mean_$RESOURCE$ value = go_t_price } }
	export_resource_stockpile_to_variable = { resource = $RESOURCE$ variable = go_t_stock }
	export_resource_maximum_to_variable  = { resource = $RESOURCE$ variable = go_t_max }
	export_resource_income_to_variable   = { resource = $RESOURCE$ variable = go_t_income }

	# per-resource status is logged by the CALLER (galactic_overhaul.2): scripted-effect
	# logs do NOT render [This.x] scope macros at runtime, but event logs do.

	# effective price = price * (1 + impact_coef * pressure)
	set_variable = { which = go_t_eff value = go_pressure_$RESOURCE$ }
	multiply_variable = { which = go_t_eff value = @go_trade_impact_coef }
	change_variable = { which = go_t_eff value = 1 }
	multiply_variable = { which = go_t_eff value = go_t_price }
	if = { limit = { check_variable = { which = go_t_eff value < 0 } } set_variable = { which = go_t_eff value = 0 } }

	# sell trigger level = cap * sell_cap_frac
	set_variable = { which = go_t_sellthresh value = go_t_max }
	multiply_variable = { which = go_t_sellthresh value = @go_trade_sell_cap_frac }

	# restock target level = cap * buy_target_frac (buy up to here when below it)
	set_variable = { which = go_t_buytarget value = go_t_max }
	multiply_variable = { which = go_t_buytarget value = $TARGETFRAC$ }

	# Opportunistic sells must never dip below the level the RESTOCK branch defends, or the
	# two halves of this effect fight: opportunistic sells at the sell fee, restock buys the
	# same units back at the buy fee next month, burning the spread every lap. Floor =
	# max(@go_opp_sell_floor, restock target).
	set_variable = { which = go_t_oppfloor value = @go_opp_sell_floor }
	if = {
		limit = { check_variable = { which = go_t_buytarget value > go_t_oppfloor } }
		set_variable = { which = go_t_oppfloor value = go_t_buytarget }
	}

	# Is the TRADE pool near its own cap? Gate for selling: don't convert a soon-to-waste
	# commodity into soon-to-waste trade. A near-full trade pool also signals there is
	# nothing to spend the proceeds on (the Reserve sweep and other restocks would have
	# drained it otherwise), which is exactly "nothing else to buy with the extra trade".
	export_resource_stockpile_to_variable = { resource = trade variable = go_t_tradenow }
	export_resource_maximum_to_variable   = { resource = trade variable = go_t_trademax }
	set_variable = { which = go_t_tradefull value = 0 }
	if = {
		limit = { check_variable = { which = go_t_trademax value > 0 } }
		set_variable = { which = go_t_tradehi value = go_t_trademax }
		multiply_variable = { which = go_t_tradehi value = @go_trade_full_frac }
		if = {
			limit = { check_variable = { which = go_t_tradenow value >= go_t_tradehi } }
			set_variable = { which = go_t_tradefull value = 1 }
		}
	}

	# ===================== SELL: near cap AND surplus income AND trade has room =====================
	if = {
		limit = {
			check_variable = { which = go_t_price value > 0 }
			check_variable = { which = go_t_max value > 0 }
			check_variable = { which = go_t_stock value >= go_t_sellthresh }
			check_variable = { which = go_t_income value > 0 }
			check_variable = { which = go_t_tradefull value = 0 }
		}
		# sell down to cap * keep_frac, capped per month
		set_variable = { which = go_t_keep value = go_t_max }
		multiply_variable = { which = go_t_keep value = @go_trade_sell_keep_frac }
		set_variable = { which = go_t_amt value = go_t_stock }
		subtract_variable = { which = go_t_amt value = go_t_keep }
		if = {
			limit = { check_variable = { which = go_t_amt value > $MAXMONTH$ } }
			set_variable = { which = go_t_amt value = $MAXMONTH$ }
		}
		if = {
			limit = { check_variable = { which = go_t_amt value >= @go_trade_min_amt } }
			set_variable = { which = go_t_gain value = go_t_amt }
			multiply_variable = { which = go_t_gain value = go_t_eff }
			multiply_variable = { which = go_t_gain value = @go_trade_sell_feemult }
			add_resource = { $RESOURCE$ = -1 mult = go_t_amt }
			add_resource = { trade = 1 mult = go_t_gain }
			subtract_variable = { which = go_pressure_$RESOURCE$ value = go_t_amt }
			set_variable = { which = go_t_action value = 1 }
		}
	}
	# ===================== BUY: restock toward the reserve target =====================
	# Whenever stock is below cap * buy_target_frac, buy up TOWARD that reserve so bursty
	# consumers (alloys, especially) don't run dry - independent of income sign.
	# Amount bought = min( target - stock, per-month cap, what spare trade can afford ).
	else_if = {
		limit = {
			check_variable = { which = go_t_price value > 0 }
			check_variable = { which = go_t_max value > 0 }
			check_variable = { which = go_t_stock value < go_t_buytarget }
		}
		# how far below the reserve we are = the amount we want to buy
		set_variable = { which = go_t_amt value = go_t_buytarget }
		subtract_variable = { which = go_t_amt value = go_t_stock }
		# rate limit per month
		if = {
			limit = { check_variable = { which = go_t_amt value > $MAXMONTH$ } }
			set_variable = { which = go_t_amt value = $MAXMONTH$ }
		}
		# spare trade above the reserve, and the per-unit cost
		export_resource_stockpile_to_variable = { resource = trade variable = go_t_energy }
		subtract_variable = { which = go_t_energy value = @go_trade_energy_floor }
		set_variable = { which = go_t_unit value = go_t_eff }
		multiply_variable = { which = go_t_unit value = @go_trade_buy_feemult }
		if = {
			limit = {
				check_variable = { which = go_t_energy value > 0 }
				check_variable = { which = go_t_unit value > 0 }
			}
			# cap the buy by what the spare trade can actually afford
			# Fair share: cap this resource's spend at a fraction of the spare trade, so the
			# FIRST resource processed can't consume the entire budget. In the 2026-07-26
			# playtest energy (processed first) took the spare trade and alloys sat at 195
			# of a 35000 cap with act=0 all game, despite being far below its restock target.
			multiply_variable = { which = go_t_energy value = @go_trade_share_frac }
			set_variable = { which = go_t_afford value = go_t_energy }
			divide_variable = { which = go_t_afford value = go_t_unit }
			if = {
				limit = { check_variable = { which = go_t_amt value > go_t_afford } }
				set_variable = { which = go_t_amt value = go_t_afford }
			}
			if = {
				limit = { check_variable = { which = go_t_amt value >= @go_trade_min_amt } }
				set_variable = { which = go_t_cost value = go_t_amt }
				multiply_variable = { which = go_t_cost value = go_t_unit }
				add_resource = { $RESOURCE$ = 1 mult = go_t_amt }
				add_resource = { trade = -1 mult = go_t_cost }
				change_variable = { which = go_pressure_$RESOURCE$ value = go_t_amt }
				set_variable = { which = go_t_action value = 2 }
			}
		}
	}

	# ===================== OPPORTUNISTIC: buy cheap / sell dear vs a price EMA =====================
	# Only for the Opportunistic desk, and only if the cap/income safety logic did NOT already act.
	# Tracks a per-resource price EMA (go_opp_mean_$RESOURCE$); buys below buy_mult*mean, sells above
	# sell_mult*mean (keeping a reserve). act codes: 3 = opportunistic buy, 4 = opportunistic sell.
	if = {
		limit = {
			has_policy_flag = go_trade_opportunistic
			check_variable = { which = go_t_action value = 0 }
			check_variable = { which = go_t_price value > 0 }
		}
		set_variable = { which = go_t_lo value = go_opp_mean_$RESOURCE$ }
		multiply_variable = { which = go_t_lo value = @go_opp_buy_mult }
		set_variable = { which = go_t_hi value = go_opp_mean_$RESOURCE$ }
		multiply_variable = { which = go_t_hi value = @go_opp_sell_mult }

		# --- BUY when cheap (price < lo) and still below the cap ---
		if = {
			limit = {
				check_variable = { which = go_t_price value < go_t_lo }
				check_variable = { which = go_t_stock value < go_t_max }
			}
			export_resource_stockpile_to_variable = { resource = trade variable = go_t_energy }
			subtract_variable = { which = go_t_energy value = @go_trade_energy_floor }
			set_variable = { which = go_t_unit value = go_t_eff }
			multiply_variable = { which = go_t_unit value = @go_opp_buy_feemult }
			if = {
				limit = { check_variable = { which = go_t_energy value > 0 } check_variable = { which = go_t_unit value > 0 } }
				set_variable = { which = go_t_amt value = go_t_energy }
				divide_variable = { which = go_t_amt value = go_t_unit }
				if = { limit = { check_variable = { which = go_t_amt value > $MAXMONTH$ } } set_variable = { which = go_t_amt value = $MAXMONTH$ } }
				# don't buy past the storage cap - the overflow would be wasted outright
				set_variable = { which = go_t_head value = go_t_max }
				subtract_variable = { which = go_t_head value = go_t_stock }
				if = { limit = { check_variable = { which = go_t_amt value > go_t_head } } set_variable = { which = go_t_amt value = go_t_head } }
				if = {
					limit = { check_variable = { which = go_t_amt value >= @go_trade_min_amt } }
					set_variable = { which = go_t_cost value = go_t_amt }
					multiply_variable = { which = go_t_cost value = go_t_unit }
					add_resource = { $RESOURCE$ = 1 mult = go_t_amt }
					add_resource = { trade = -1 mult = go_t_cost }
					change_variable = { which = go_pressure_$RESOURCE$ value = go_t_amt }
					# running P&L of the active desk (trade out now, back in on the sell leg)
					if = { limit = { NOT = { is_variable_set = go_opp_pnl } } set_variable = { which = go_opp_pnl value = 0 } }
					subtract_variable = { which = go_opp_pnl value = go_t_cost }
					set_variable = { which = go_t_action value = 3 }
				}
			}
		}
		# --- SELL when dear (price > hi) and we hold above the reserve floor ---
		else_if = {
			limit = {
				check_variable = { which = go_t_price value > go_t_hi }
				check_variable = { which = go_t_stock value > go_t_oppfloor }
				check_variable = { which = go_t_tradefull value = 0 }
			}
			set_variable = { which = go_t_amt value = go_t_stock }
			subtract_variable = { which = go_t_amt value = go_t_oppfloor }
			if = { limit = { check_variable = { which = go_t_amt value > $MAXMONTH$ } } set_variable = { which = go_t_amt value = $MAXMONTH$ } }
			if = {
				limit = { check_variable = { which = go_t_amt value >= @go_trade_min_amt } }
				set_variable = { which = go_t_gain value = go_t_amt }
				multiply_variable = { which = go_t_gain value = go_t_eff }
				multiply_variable = { which = go_t_gain value = @go_opp_sell_feemult }
				add_resource = { $RESOURCE$ = -1 mult = go_t_amt }
				add_resource = { trade = 1 mult = go_t_gain }
				subtract_variable = { which = go_pressure_$RESOURCE$ value = go_t_amt }
				# running P&L of the active desk
				if = { limit = { NOT = { is_variable_set = go_opp_pnl } } set_variable = { which = go_opp_pnl value = 0 } }
				change_variable = { which = go_opp_pnl value = go_t_gain }
				set_variable = { which = go_t_action value = 4 }
			}
		}

		clear_variable = go_t_lo
		clear_variable = go_t_hi
		clear_variable = go_t_tmp
	}

	# go_t_action/amt/stock/max/income/price left set for galactic_overhaul.2 to log, cleared there
	# Maintain the reference EMA unconditionally, AFTER this month's decision so the
	# decision used the prior mean: mean += alpha * (price - mean).
	if = {
		limit = { check_variable = { which = go_t_price value > 0 } }
		set_variable = { which = go_t_tmp value = go_t_price }
		subtract_variable = { which = go_t_tmp value = go_opp_mean_$RESOURCE$ }
		multiply_variable = { which = go_t_tmp value = @go_opp_alpha }
		change_variable = { which = go_opp_mean_$RESOURCE$ value = go_t_tmp }
	}

	clear_variable = go_t_eff
	clear_variable = go_t_sellthresh
	clear_variable = go_t_buytarget
	clear_variable = go_t_keep
	clear_variable = go_t_gain
	clear_variable = go_t_cost
	clear_variable = go_t_energy
	clear_variable = go_t_unit
	clear_variable = go_t_afford
	clear_variable = go_t_tradenow
	clear_variable = go_t_trademax
	clear_variable = go_t_tradehi
	clear_variable = go_t_tradefull
	clear_variable = go_t_head
	clear_variable = go_t_oppfloor
}
